'use client';
import React, { useEffect, useState, useMemo, useRef, useCallback } from 'react';
import { useProgress } from '@react-three/drei';

export function LoadingScreen() {
  const { progress } = useProgress();
  const [show, setShow] = useState(true);

  // Use refs instead of state for high-frequency updates to avoid React re-renders
  // that compete with Three.js texture loading on the main thread.
  const delayedProgressRef = useRef(0);
  const phraseIndexRef = useRef(0);
  const rafIdRef = useRef<number>(0);
  const phraseIntervalRef = useRef<ReturnType<typeof setInterval>>(null);
  const hasDispatchedLoadedRef = useRef(false);
  const fadeTimerRef = useRef<ReturnType<typeof setTimeout>>(null);

  // DOM refs for direct manipulation (bypass React reconciliation)
  const containerRef = useRef<HTMLDivElement>(null);
  const percentRef = useRef<HTMLDivElement>(null);
  const phraseRefs = useRef<(HTMLParagraphElement | null)[]>([]);

  const loadingPhrases = useMemo(() => [
    "Preparing beautiful memories...",
    "Menyiapkan hari bahagia...",
    "Menyusun setiap detail kisah...",
    "Sesaat lagi...",
    "Symphony of love is tuning..."
  ], []);

  // Store progress in a ref so the rAF loop always sees the latest value
  const progressRef = useRef(progress);
  progressRef.current = progress;

  // Phrase cycling via direct DOM class toggling (no React re-render)
  useEffect(() => {
    phraseIntervalRef.current = setInterval(() => {
      const prev = phraseIndexRef.current;
      const next = (prev + 1) % loadingPhrases.length;
      phraseIndexRef.current = next;

      // Directly toggle visibility classes on phrase elements
      phraseRefs.current.forEach((el, idx) => {
        if (!el) return;
        if (idx === next) {
          el.className = 'absolute text-[11px] font-serif italic text-[#A8B8A6] tracking-wider transition-all duration-700 ease-in-out opacity-100 translate-y-0 scale-100';
        } else {
          el.className = 'absolute text-[11px] font-serif italic text-[#A8B8A6] tracking-wider transition-all duration-700 ease-in-out opacity-0 translate-y-4 scale-95 pointer-events-none';
        }
      });
    }, 2800);
    return () => {
      if (phraseIntervalRef.current) clearInterval(phraseIntervalRef.current);
    };
  }, [loadingPhrases.length]);

  // Main animation loop using requestAnimationFrame — never blocks the main thread
  // and naturally yields to browser layout/paint work unlike setInterval+setState.
  const animate = useCallback(() => {
    const currentProgress = progressRef.current;
    let dp = delayedProgressRef.current;

    if (dp < currentProgress) {
      dp = Math.min(dp + 1.2, currentProgress);
    } else if (currentProgress === 100 && dp < 100) {
      dp = Math.min(dp + 1.5, 100);
    }

    delayedProgressRef.current = dp;

    // Direct DOM updates — zero React overhead
    if (percentRef.current) {
      percentRef.current.textContent = `${Math.floor(dp)}%`;
    }

    // Handle fade-out and completion
    if (dp >= 100) {
      if (containerRef.current && !containerRef.current.classList.contains('opacity-0')) {
        containerRef.current.classList.remove('opacity-100');
        containerRef.current.classList.add('opacity-0');
      }

      if (!hasDispatchedLoadedRef.current) {
        hasDispatchedLoadedRef.current = true;
        window.dispatchEvent(new Event('weddingLoaded'));

        fadeTimerRef.current = setTimeout(() => {
          setShow(false); // Single React state update — unmounts the component
        }, 1200);
      }
    }

    rafIdRef.current = requestAnimationFrame(animate);
  }, []);

  useEffect(() => {
    rafIdRef.current = requestAnimationFrame(animate);
    return () => {
      cancelAnimationFrame(rafIdRef.current);
      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    };
  }, [animate]);

  if (!show) return null;

  return (
    <div 
      ref={containerRef}
      className="fixed inset-0 z-[999999] flex flex-col items-center justify-center bg-[#0A120D]/65 backdrop-blur-[6px] transition-opacity duration-1000 opacity-100 pointer-events-none"
      style={{ transitionTimingFunction: 'cubic-bezier(0.16, 1, 0.3, 1)' }}
    >
      {/* Premium Ambient Background Glows matching the wedding config */}
      <div className="absolute top-1/4 left-1/4 w-96 h-96 rounded-full bg-[#4B182B]/20 blur-[120px]" />
      <div className="absolute bottom-1/4 right-1/4 w-96 h-96 rounded-full bg-[#A8B8A6]/10 blur-[120px]" />

      {/* Loader Container */}
      <div className="relative flex flex-col items-center max-w-sm px-6 text-center select-none">
        
        {/* Circular Progress Ring with Monogram */}
        <div className="relative w-40 h-40 flex items-center justify-center mb-8">
          
          {/* Central Monogram (Main Selection Color + Sage Green Subtext) */}
          {/* <div className="flex flex-col items-center justify-center z-10">
            <span className="text-[#4B182B] font-serif text-3xl font-light tracking-widest mt-1">Q & B</span>
            <span className="text-[#A8B8A6] font-serif text-[10px] tracking-[0.25em] uppercase mt-2">Wedding</span>
          </div> */}

          {/* Percentage Indicator Badge (Selection background + Ivory text) */}
          <div
            ref={percentRef}
            className="absolute bottom-0 text-[11px] font-sans tracking-widest text-[#F8F3ED] bg-[#4B182B] px-3 py-0.5 border border-[#4B182B]/35 rounded-full shadow-lg shadow-[#4B182B]/10"
          >
            0%
          </div>
        </div>

        {/* Title & Romantic Subtitles */}
        <h1 className="text-[#F8F3ED] font-serif text-lg tracking-[0.2em] uppercase font-light mb-3">
          Qonita &amp; Bagja
        </h1>

        {/* Dynamic Glowing Text Cross-fader (Sage Subtext Color) */}
        <div className="h-6 overflow-hidden relative w-64 flex items-center justify-center">
          {loadingPhrases.map((phrase, idx) => (
            <p
              key={idx}
              ref={el => { phraseRefs.current[idx] = el; }}
              className={`absolute text-[11px] font-serif italic text-[#A8B8A6] tracking-wider transition-all duration-700 ease-in-out ${
                idx === 0 
                  ? 'opacity-100 translate-y-0 scale-100' 
                  : 'opacity-0 translate-y-4 scale-95 pointer-events-none'
              }`}
            >
              &quot;{phrase}&quot;
            </p>
          ))}
        </div>
      </div>
      
      {/* Bottom copyright/visual ornament */}
      <div className="absolute bottom-8 flex flex-col items-center opacity-30 tracking-[0.4em] uppercase text-[8px] text-[#F8F3ED]">
        <span>Loading Experience</span>
      </div>
    </div>
  );
}
